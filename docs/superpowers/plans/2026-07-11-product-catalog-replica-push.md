# Product Catalog Replica (push model) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Keep a per-org Postgres replica of the 1C product catalog — fed by 1C pushing into our API — so barcode scan and a new name search are instant and survive 1C outages, while stock/price stay live and images stream through a backend proxy.

**Architecture:** 1C authenticates with a per-org `webhook_token` and pushes catalog batches + explicit delete events into two DRF ingest endpoints (no worker, no queue). Scan reads catalog from the replica and fetches only live stock from 1C; images are proxied (fetch-with-Basic-auth, org-scoped, never mirrored). Name search is an org-scoped Postgres trigram query.

**Tech Stack:** Python 3.13, Django 6, DRF, PostgreSQL 17 (SQLite dev fallback), `httpx`, Fernet; React 18 + Ant Design frontend.

## Global Constraints

- **Migrations are checked in and additive** — generate with `makemigrations`, never hand-edit an *applied* migration; prefer additive changes (`backend/core/migrations/`).
- **Multi-tenancy is two-layer** — every read view declares a permission class **and** filters its queryset by `request.user.organization`. Both must agree or data leaks across orgs.
- **Error envelope** — machine-branchable errors use `{"code": "...", "detail": "...", ...}` (see `_consult_error_response`, `backend/core/views.py:129`).
- **drf-spectacular** — every new view is decorated `@extend_schema(tags=[...])`; add new tags to `SPECTACULAR_SETTINGS['TAGS']` in `backend/settings.py`.
- **Ingest tenancy invariant** — the org is derived from the push token only; request bodies never name an org.
- **Image proxy invariant** — the proxy fetches only URLs already stored on the requested product row for the caller's org; it never fetches a client-supplied URL (no SSRF).
- **Test runner** — Django, not pytest: `cd backend && python manage.py test core.tests.<Class>.<method>`. Test DB is SQLite; `pg_trgm` is unavailable there, so name search must fall back to `icontains`.
- **FERNET_KEY** must be set in the environment for any test that calls `decrypt_password()`.

        constraints = [models.UniqueConstraint(fields=["organization", "sku"], name="uq_product_org_sku")]

    def __str__(self):

    product = models.ForeignKey(Product, on_delete=models.CASCADE, related_name="barcodes")
    barcode = models.CharField(max_length=255, db_index=True)

        constraints = [models.UniqueConstraint(fields=["product", "barcode"], name="uq_barcode_per_product")]

    organization = models.OneToOneField(Organization, on_delete=models.CASCADE, related_name="catalog_ingest_state")
    last_full_push_at = models.DateTimeField(null=True, blank=True)
    last_delta_push_at = models.DateTimeField(null=True, blank=True)
    last_delete_at = models.DateTimeField(null=True, blank=True)
    status = models.CharField(max_length=16, default="ok")  # ok | stale | error
    last_error = models.TextField(blank=True, default="")
    received = models.PositiveIntegerField(default=0)      # counts of the last push
    upserted = models.PositiveIntegerField(default=0)
    deactivated = models.PositiveIntegerField(default=0)
    images_failed = models.PositiveIntegerField(default=0)

    def is_stale(self) -> bool:

def add_trgm(apps, schema_editor):
    if schema_editor.connection.vendor != "postgresql":
        return
    schema_editor.execute("CREATE EXTENSION IF NOT EXISTS pg_trgm")
    schema_editor.execute(

def drop_trgm(apps, schema_editor):
    if schema_editor.connection.vendor != "postgresql":

"""Pure helpers for the catalog replica — no Django model imports, easy to unit-test."""
import hashlib
import json


def row_hash(product: dict) -> str:
    """Stable sha256 over the synced fields; barcode order does not matter."""
    payload = json.dumps(
        {
            "article": product.get("article") or "",
            "name": product.get("name") or "",
            "price": str(product.get("price") if product.get("price") is not None else ""),
            "barcodes": sorted(product.get("barcodes") or []),
            "image_urls": product.get("image_urls") or [],
        },
        sort_keys=True, ensure_ascii=False,
    )
    return hashlib.sha256(payload.encode("utf-8")).hexdigest()


def proxy_image_paths(sku: str, count: int) -> list[str]:
    """Relative proxy paths the frontend loads instead of raw 1C URLs."""
    return [f"catalog/products/{sku}/image/{i}/" for i in range(count)]
```

"""Per-org push-token auth. The org is derived ENTIRELY from the token — request
bodies never name an org — so a push can only write the token-owner's rows."""
import hmac

from rest_framework.exceptions import AuthenticationFailed

from .models import Organization


def _extract_token(request) -> str:
    header = request.headers.get("X-Webhook-Token")
    if header:
        return header.strip()
    auth = request.headers.get("Authorization", "")
    if auth.startswith("Bearer "):
        return auth[len("Bearer "):].strip()
    return ""


def organization_from_push(request) -> Organization:
    token = _extract_token(request)
    if not token:
        raise AuthenticationFailed("Missing push token.")
    # Unique index makes this a single-row hit; constant-time compare avoids a timing oracle.
    org = Organization.objects.filter(webhook_token=token).first()
    if org is None or not hmac.compare_digest(org.webhook_token, token):
        raise AuthenticationFailed("Invalid push token.")
    return org
```

```python
@extend_schema(tags=["Catalog Ingest"])
class CatalogProductIngestAPIView(APIView):
    permission_classes = []  # authenticated by per-org push token, not JWT
    http_method_names = ["post"]

    def post(self, request: Request) -> Response:
        org = organization_from_push(request)  # raises AuthenticationFailed on bad/missing token
        products = request.data.get("products") or []
        is_full = bool(request.data.get("is_full"))
        upserted = skipped = 0

        with transaction.atomic():
            for item in products:
                sku = item.get("sku")
                if not sku:
                    continue
                new_hash = row_hash(item)
                existing = Product.objects.filter(organization=org, sku=sku).first()
                if existing and existing.row_hash == new_hash and existing.is_active:
                    skipped += 1
                    continue
                obj, _ = Product.objects.update_or_create(
                    organization=org, sku=sku,
                    defaults={
                        "article": item.get("article") or "",
                        "name": item.get("name") or "",
                        "price": item.get("price"),
                        "image_urls": item.get("image_urls") or [],
                        "row_hash": new_hash,
                        "is_active": True,
                        "deactivated_at": None,
                        "pushed_at": timezone.now(),
                    },
                )
                obj.barcodes.all().delete()
                ProductBarcode.objects.bulk_create(
                    [ProductBarcode(product=obj, barcode=b) for b in (item.get("barcodes") or [])]
                )
                upserted += 1

            state, _ = CatalogIngestState.objects.get_or_create(organization=org)
            now = timezone.now()
            if is_full:
                state.last_full_push_at = now
            else:
                state.last_delta_push_at = now
            state.received, state.upserted, state.status, state.last_error = len(products), upserted, "ok", ""
            state.save()

        return Response({"received": len(products), "upserted": upserted, "skipped": skipped})
```

    http_method_names = ["get"]

    def get(self, request: Request) -> Response:
        q = (request.query_params.get("q") or "").strip()
        if not q:

        qs = Product.objects.filter(organization=request.user.organization, is_active=True)
        if connection.vendor == "postgresql":

            qs = qs.annotate(rank=TrigramSimilarity("name", q)).filter(rank__gt=0.1).order_by("-rank")

                "image": proxy_image_paths(p.sku, len(p.image_urls))[0] if p.image_urls else None,
            }
            for p in qs[:20]
        ]
        return Response(CatalogProductSerializer(rows, many=True).data)

}
